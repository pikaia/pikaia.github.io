Walk around Tiong Bahru today and the rounded corners, curved balconies and porthole windows read as a heritage flex — a well-preserved pocket of 1930s Art Deco that estate agents now market as a lifestyle. Almost none of it was built to be charming. It was Singapore's first public housing, built by a colonial body that no longer exists, to get people out of exactly the kind of overcrowded, disease-ridden slum that Tiong Bahru itself had once been.

  function splitLongSentence(text, offset, duration) {
    if (text.length <= MAX_CHARS) return [{ text: text, offset_s: offset, duration_s: duration }];
    var parts = text.split(/(?<=[:;])\s+/);
    var chunks = [];
    parts.forEach(function (p) {
      if (p.length <= MAX_CHARS) { chunks.push(p); return; }
      var words = p.split(' ');
      var cur = '';
      words.forEach(function (w) {
        if ((cur + ' ' + w).trim().length > MAX_CHARS) { chunks.push(cur.trim()); cur = w; }
        else { cur = (cur + ' ' + w).trim(); }
      });
      if (cur) chunks.push(cur.trim());
    });
    var totalChars = chunks.reduce(function (a, c) { return a + c.length; }, 0);
    var t = offset;
    return chunks.map(function (c) {
      var share = duration * (c.length / totalChars);
      var entry = { text: c, offset_s: t, duration_s: share };
      t += share;
      return entry;
    });
  }
  var captionChunks = [];
  sentences.forEach(function (s) {
    captionChunks = captionChunks.concat(splitLongSentence(s.text, s.offset_s, s.duration_s));
  });

  var slideDurations = imageSchedule.map(function (entry, i) {
    var next = imageSchedule[i + 1];
    return (next ? next.t : TOTAL_DURATION) - entry.t;
  });

  var viewer = document.getElementById('watch-viewer');
  var stage = document.getElementById('watch-stage');
  var captionEl = document.getElementById('watch-caption');
  var closeBtn = document.getElementById('watch-close');
  var playBtn = document.getElementById('watch-play');
  var progress = document.getElementById('watch-progress');
  var progressFill = document.getElementById('watch-progress-fill');
  var timeEl = document.getElementById('watch-time');
  var watchWidget = document.getElementById('watch-widget');

  // Separate Audio object from the Listen widget's <audio> element -
  // never share the `audio` variable across the two IIFEs (the #1
  // recurring bug in this pipeline's history, see
  // docs/production-pipeline.md section 13).
  var listenAudio = document.getElementById('listen-audio');
  var watchAudio = new Audio(listenAudio.querySelector('source').src);
  watchAudio.preload = 'none';

  var styleEl = document.createElement('style');
  document.head.appendChild(styleEl);

  var slideEls = slides.map(function (s, i) {
    var el = document.createElement('div');
    el.style.cssText = 'position:absolute;inset:0;opacity:0;transition:opacity 0.8s ease;';
    var ease = s.ease || 'linear';
    var dur = slideDurations[imageSchedule.findIndex(function (e) { return e.slide === i; })];

    if (s.type === 'chart') {
      var chart = buildChartSlide(s);
      el.appendChild(chart.svg);
      el._chartUpdate = chart.update;
    } else if (s.type === 'letterbox') {
      var bg = document.createElement('div');
      bg.style.cssText = 'position:absolute;inset:-8%;background-size:cover;background-position:center;filter:blur(30px) brightness(0.55);background-image:url(\'' + s.src + '\');';
      bg.style.animation = 'kb' + i + ' ' + dur + 's ' + ease + ' forwards';
      var fg = document.createElement('div');
      fg.style.cssText = 'position:absolute;inset:6%;background-size:contain;background-position:center;background-repeat:no-repeat;background-image:url(\'' + s.src + '\');';
      el.appendChild(bg); el.appendChild(fg);
      el._animTargets = [bg];
    } else {
      var layer = document.createElement('div');
      layer.style.cssText = 'position:absolute;inset:-8%;background-size:cover;background-image:url(\'' + s.src + '\');';
      layer.style.animation = 'kb' + i + ' ' + dur + 's ' + ease + ' forwards';
      el.appendChild(layer);
      el._animTargets = [layer];
    }
    if (s.type !== 'chart') {
      styleEl.textContent += '@keyframes kb' + i + ' { 0% { transform: scale(' + s.zoom[0] + '); background-position: ' + s.pan[0] + '; } 50% { transform: scale(' + s.zoom[1] + '); background-position: ' + s.pan[1] + '; } 100% { transform: scale(' + s.zoom[2] + '); background-position: ' + s.pan[2] + '; } }\n';
    }
    stage.appendChild(el);
    return el;
  });
  slides.forEach(function (s) { if (s.src) { var img = new Image(); img.src = s.src; } });

  var currentIndex = -1, currentSentenceIndex = -1;
  function fmtTime(t) {
    if (!isFinite(t)) return '0:00';
    var m = Math.floor(t / 60), s = Math.floor(t % 60);
    return m + ':' + (s < 10 ? '0' : '') + s;
  }
  function slideIndexForTime(t) {
    var idx = 0;
    for (var i = 0; i < imageSchedule.length; i++) { if (imageSchedule[i].t <= t) idx = imageSchedule[i].slide; else break; }
    return idx;
  }
  function sentenceIndexForTime(t) {
    var idx = 0;
    for (var i = 0; i < captionChunks.length; i++) { if (captionChunks[i].offset_s <= t) idx = i; else break; }
    return idx;
  }
  function updateForTime(t) {
    var sIdx = slideIndexForTime(t);
    if (sIdx !== currentIndex) {
      if (currentIndex >= 0) slideEls[currentIndex].style.opacity = '0';
      slideEls[sIdx].style.opacity = '1';
      currentIndex = sIdx;
    }
    if (slideEls[sIdx]._chartUpdate) {
      var schedIdx = imageSchedule.findIndex(function (e) { return e.slide === sIdx; });
      var segStart = imageSchedule[schedIdx].t;
      var nextEntry = imageSchedule[schedIdx + 1];
      var segEnd = nextEntry ? nextEntry.t : TOTAL_DURATION;
      var chartProgress = Math.min(1, Math.max(0, (t - segStart) / Math.max(0.001, segEnd - segStart)));
      slideEls[sIdx]._chartUpdate(t, chartProgress);
    }
    var cIdx = sentenceIndexForTime(t);
    if (cIdx !== currentSentenceIndex) { captionEl.textContent = captionChunks[cIdx].text; currentSentenceIndex = cIdx; }
    var pct = Math.min(100, (t / TOTAL_DURATION) * 100);
    progressFill.style.width = pct + '%';
    timeEl.textContent = fmtTime(t) + ' / ' + fmtTime(TOTAL_DURATION);
  }
  function openViewer() {
    viewer.style.display = 'block';
    currentIndex = -1; currentSentenceIndex = -1;
    slideEls.forEach(function (el) { el.style.opacity = '0'; });
    watchAudio.currentTime = 0;
    updateForTime(0);
    watchAudio.play();
  }
  function closeViewer() { viewer.style.display = 'none'; watchAudio.pause(); }

  watchWidget.addEventListener('click', openViewer);
  watchWidget.addEventListener('keydown', function (e) { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); openViewer(); } });
  closeBtn.addEventListener('click', closeViewer);
  playBtn.addEventListener('click', function () { if (watchAudio.paused) watchAudio.play(); else watchAudio.pause(); });
  watchAudio.addEventListener('play', function () { playBtn.innerHTML = '&#10074;&#10074;'; });
  watchAudio.addEventListener('pause', function () { playBtn.innerHTML = '&#9654;'; });
  watchAudio.addEventListener('timeupdate', function () { updateForTime(watchAudio.currentTime); });
  watchAudio.addEventListener('ended', closeViewer);
  progress.addEventListener('click', function (e) {
    var rect = progress.getBoundingClientRect();
    var pct = (e.clientX - rect.left) / rect.width;
    watchAudio.currentTime = pct * TOTAL_DURATION;
  });
})();
</script>

## A slum next to the hospital

By the 1910s and 20s, the ground that would become Tiong Bahru was a byword for exactly the conditions colonial Singapore was trying to be seen tackling. The area sat close to the General Hospital, which made its state a particular embarrassment: mangrove swamp, poor drainage, and roads the *Singapore Free Press* in 1914 described as "dilapidated and dangerous" and "in a disgraceful state of neglect, being full of huge holes and ruts." It bred mosquitoes and disease on a hospital's doorstep. Further off in Chinatown, the same overcrowding was worse and better documented: a 1907 public health report and a 1918 housing commission had already argued that Singapore's slums needed more than road tax and ad hoc clearance — they needed a dedicated body with the power to plan, build and rehouse. The municipal health officer P. S. Hunter's own study of Chinatown's congestion made the same case directly to the government.

## The Trust

That body was the Singapore Improvement Trust, established on 1 July 1927 under the Singapore Improvement Ordinance, empowered "to provide for the Improvement of the Town and Island of Singapore." Tiong Bahru was its proving ground. A clearance scheme began in 1925, land — 33 hectares of it — was formally acquired in 1926 for around $600,000, and basic infrastructure (roads, drains, a canal) took until 1931 to complete, at a further $1.5 million. Only then did the Trust start building flats: construction began in March 1936, and the first completed block — 28 units over four shops — opened that December.

By 1941 the Trust had put up 784 flats housing more than 6,000 people. The architects — Lincoln Page, Robert F. N. Kan and A. G. Church — borrowed from British new towns like Stevenage, Harlow and Crawley, but built in a Streamline Moderne style: rounded corners, sweeping horizontal lines, porthole windows, stainless-steel railings, spiral staircases, and prewar blocks that folded in shophouse-style courtyards and air-wells. Locals gave the earliest blocks a nickname of their own — *puay kee chu*, "aeroplane houses" — because their curved, tower-like massing reminded people of Kallang Airport's control tower, finished around the same time on the other side of the island.

## The den of beauties

Tiong Bahru's flats were a step up from a shophouse cubicle, and the people who could afford that step up were not always who a government housing scheme had in mind. The estate's Chinese nickname, *mei ren wo* — "den of beauties" — came from its reputation for housing the mistresses of wealthy men, along with cabaret singers and dance hostesses from the nightclub scene around Great World Amusement Park nearby. Locals had a third name for it, only half joking: the "Hollywood of Singapore." The flats themselves didn't discriminate — Chinese, Indian, Eurasian and European families lived there too — but the nickname stuck to the estate for decades.

## War, then a handoff

The Japanese Occupation damaged rooftops across the estate and stalled further building, though it didn't stop people moving in: close to 40% of Tiong Bahru's postwar residents had actually taken up their flats during the Occupation itself. The community that formed there kept growing afterward — the first residents' club appeared in 1948, and Singapore's first community centre opened in Tiong Bahru in 1951. But the Trust that had built it all was on borrowed time. The Housing and Development Ordinance took effect on 1 February 1960, formally dissolving the Singapore Improvement Trust and handing its housing mandate to a new statutory board — the Housing and Development Board. By then the Trust had also drawn up Singapore's 1952–58 Master Plan, the zoning and new-town framework HDB would build its much larger housing programme on top of. Tiong Bahru wasn't just SIT's biggest project; it was the prototype for everything HDB did next.

## From decline to heritage

For a while, Tiong Bahru just aged. Rental flats were converted to sales in the 1960s and 70s, the last of the surrounding slums were cleared by the mid-1970s, and by the 1980s and 90s the estate had an ageing population and little else to recommend it — small, dated flats with none of the newer HDB towns' scale or amenities. What reversed that was the opposite of what built it: not need, but heritage. Twenty prewar blocks were given conservation status in 2003, and from the early 2000s young professionals, gallery owners and boutique operators moved in, drawn by exactly the Art Deco character the original tenants had never been asked to appreciate. Property prices climbed with them.

![Tiong Bahru Market, a curved red-and-white building with an escalator and a sign for the food centre](https://upload.wikimedia.org/wikipedia/commons/thumb/1/1b/Tiong_Bahru_Market%2C_exterior.jpg/1280px-Tiong_Bahru_Market%2C_exterior.jpg)

*Tiong Bahru Market today — rebuilt, but still tracing the estate's original curved footprint, now surrounded by cafes and boutiques rather than the SIT-era shops it replaced. (Photo: Kbseah, CC BY-SA 4.0, via Wikimedia Commons.)*

Conservation flats that once housed dockworkers and cabaret girls now command some of the highest resale and rental prices of any HDB-era housing in Singapore, some converted into boutique hotels — despite carrying some of the shortest remaining leases in the public housing stock. The prewar blocks were privatised on 99-year leases dated from 1 January 1967, expiring in 2066; the postwar blocks that make up most of the 2003 conservation area run on 99-year leases from 1973, expiring in 2072. Conservation status forecloses the usual way an ageing HDB estate resets that clock — en-bloc redevelopment — so short of a government buyback and fresh lease, Tiong Bahru's remaining lease only gets shorter. In a market where remaining lease is normally one of the strongest drags on price, an estate this far into a shrinking, non-resettable lease still commanding a premium is a genuine exception, not just a high price. The estate built to solve a slum problem is today priced well out of reach of the kind of household it was designed for.

[See more photographs related to this post →](/gallery/before-hdb-singapore-improvement-trust-tiong-bahru/)

**Why it matters today:** Singapore's public housing story is usually told starting with HDB and the 1960s. Tiong Bahru is the reminder that the model — clear the slum, build a planned estate, hand it to the people who need it most — was already three decades old by the time HDB existed, tested and refined by a colonial trust nobody remembers today. The estate's second life, as heritage property for people who don't need public housing at all, is a different story the SIT could never have planned for.
